# Also usable as a standalone, SDK-free contract test project.
cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(media_core_contract LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(MEDIA_FFMPEG_FULL OFF)
    set(MEDIA_RAW_FRAME_BUFFER OFF)
    set(BUILD_TESTS OFF)
    get_filename_component(EMBEDDED_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
    add_subdirectory("${EMBEDDED_SOURCE_ROOT}/modules/media" media)
    enable_testing()
endif()
add_executable(test_media_core test_media_core.cpp)
target_link_libraries(test_media_core PRIVATE Media::Core)
add_test(NAME test_media_core COMMAND test_media_core)
if(MEDIA_FFMPEG_FULL)
    media_test(test_media_packet_buffer Media::Encoder Media::Decoder)
endif()
