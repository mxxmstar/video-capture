cmake_minimum_required(VERSION 3.21)
include(cmake/Platform.cmake)
project(embedded_device VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(cmake/BuildOptions.cmake)
include(cmake/RuntimeDependencies.cmake)

# Each platform owns dependency discovery and backend selection.
include("${EMBEDDED_PLATFORM_DIR}/CMakeLists.txt")
add_subdirectory(src/common)
if(BUILD_MEDIA)
    add_subdirectory(src/media)
endif()
if(BUILD_RENDER)
    add_subdirectory(src/render)
endif()

add_executable(execution src/main.cpp)
target_link_libraries(execution PRIVATE common_lib)
if(BUILD_RENDER)
    target_link_libraries(execution PRIVATE render_lib)
endif()
copy_runtime_dependencies(execution)

if(BUILD_TESTS)
    enable_testing()
    add_subdirectory(test)
endif()
