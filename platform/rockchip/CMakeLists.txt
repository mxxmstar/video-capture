option(ROCKCHIP_WITH_MPP "Link Rockchip Media Process Platform" ON)
option(ROCKCHIP_WITH_RGA "Link Rockchip Raster Graphic Acceleration" OFF)
function(rockchip_import name header library)
    string(TOUPPER "${name}" _key)
    find_path(ROCKCHIP_${_key}_INCLUDE_DIR NAMES "${header}"
        HINTS "${ROCKCHIP_DEPS_ROOT}/include" PATH_SUFFIXES rockchip rga)
    find_library(ROCKCHIP_${_key}_LIBRARY NAMES "${library}"
        HINTS "${ROCKCHIP_DEPS_ROOT}/lib")
    if(NOT ROCKCHIP_${_key}_INCLUDE_DIR OR NOT ROCKCHIP_${_key}_LIBRARY)
        message(FATAL_ERROR "Missing target ${name}: populate the device sysroot or ${ROCKCHIP_DEPS_ROOT}. See platform/rockchip/README.md")
    endif()
    add_library(Rockchip::${name} UNKNOWN IMPORTED GLOBAL)
    set_target_properties(Rockchip::${name} PROPERTIES
        IMPORTED_LOCATION "${ROCKCHIP_${_key}_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${ROCKCHIP_${_key}_INCLUDE_DIR}")
    target_link_libraries(embedded_platform INTERFACE Rockchip::${name})
    target_compile_definitions(embedded_platform INTERFACE EMBEDDED_HAS_${_key}=1)
endfunction()
